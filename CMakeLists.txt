cmake_minimum_required(VERSION 3.22)
project(ydotool C)

set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

# Include GNU install directory module to detect where to install
# files on Linux/Unix systems (e.g., lib vs lib64)
include(GNUInstallDirs)
find_package(Intl REQUIRED)
find_program(XGETTEXT_EXECUTABLE xgettext REQUIRED)
find_program(MSGFMT_EXECUTABLE msgfmt REQUIRED)
set(YDOTOOL_LOCALEDIR "${CMAKE_INSTALL_FULL_LOCALEDIR}" CACHE PATH "Locale catalog installation directory")
add_compile_definitions(YDOTOOL_LOCALEDIR="${YDOTOOL_LOCALEDIR}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/locale")
file(GLOB PO_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/po/*.po")
foreach(PO_FILE IN LISTS PO_FILES)
	get_filename_component(LOCALE "${PO_FILE}" NAME_WE)
	set(MO_FILE "${CMAKE_CURRENT_BINARY_DIR}/locale/${LOCALE}/LC_MESSAGES/ydotool.mo")
	add_custom_command(OUTPUT "${MO_FILE}" COMMAND "${CMAKE_COMMAND}" -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/locale/${LOCALE}/LC_MESSAGES" COMMAND "${MSGFMT_EXECUTABLE}" --check --check-format -o "${MO_FILE}" "${PO_FILE}" DEPENDS "${PO_FILE}" VERBATIM)
	list(APPEND MO_FILES "${MO_FILE}")
endforeach()
add_custom_target(ydotool_catalogs ALL DEPENDS ${MO_FILES})
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/locale/" DESTINATION "${CMAKE_INSTALL_LOCALEDIR}" OPTIONAL)

if(NOT DEFINED YDOTOOL_VERSION)
    execute_process(COMMAND git describe --tags --long --always
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        RESULT_VARIABLE RC_GIT_VER OUTPUT_VARIABLE GIT_VERSION)
else()
    set(RC_GIT_VER 0)
    set(GIT_VERSION "${YDOTOOL_VERSION}")
endif()

if (${RC_GIT_VER} EQUAL 0)
    string(STRIP ${GIT_VERSION} GIT_VERSION)
    message("-- Version: " ${GIT_VERSION})
    add_definitions(-DVERSION=\"${GIT_VERSION}\")
endif()

set(SOURCE_FILES_DAEMON Daemon/ydotoold.c)
set(SOURCE_FILES_CLIENT Client/ydotool.c Client/tool_click.c Client/tool_mousemove.c Client/tool_type.c Client/tool_key.c Client/tool_stdin.c)

add_executable(ydotoold ${SOURCE_FILES_DAEMON})
target_link_libraries(ydotoold PRIVATE Intl::Intl)
install(TARGETS ydotoold DESTINATION ${CMAKE_INSTALL_BINDIR})

add_executable(ydotool ${SOURCE_FILES_CLIENT})
target_link_libraries(ydotool PRIVATE Intl::Intl)
install(TARGETS ydotool DESTINATION ${CMAKE_INSTALL_BINDIR})

add_subdirectory(Daemon)
add_subdirectory(manpage)

option(YDOTOOL_TESTS "Build isolated input test fixtures" OFF)
if(YDOTOOL_TESTS)
    add_library(c23_contract OBJECT tests/c23_contract.c)
    add_library(fake_uinput SHARED tests/fake_uinput.c)
    list(REMOVE_ITEM SOURCE_FILES_CLIENT Client/ydotool.c)
    add_executable(send_failure tests/send_failure.c ${SOURCE_FILES_CLIENT})
    add_executable(socket_paths tests/socket_paths.c ${SOURCE_FILES_CLIENT})
    target_link_libraries(send_failure PRIVATE Intl::Intl)
    target_link_libraries(socket_paths PRIVATE Intl::Intl)
    target_link_options(socket_paths PRIVATE -Wl,--wrap=connect -Wl,--wrap=open -Wl,--wrap=__open_2)
endif()
